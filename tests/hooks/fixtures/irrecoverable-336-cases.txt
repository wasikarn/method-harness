# GH #336 cases for hooks/gates/irrecoverable.sh, replayed by tests/hooks/test-gates-other.sh.
# DENY <cmd>: exit 2. ALLOW <cmd>: exit 0, empty stdout, no would_* row.
# WOULD_DENY <rule> <cmd> / WOULD_ASK <rule> <cmd>: a shadow rule (SHADOW_RULES) matches: exit 0,
# empty stdout, one would_* row naming <rule>. The test also runs every WOULD_* line against a copy
# with SHADOW_RULES emptied and expects the enforced verdict (deny / ask).
# Each DENY and WOULD_DENY shape was run in /bin/bash 3.2, bash 5, /bin/dash and /bin/zsh against a logging stub.
#
# controls (deny on develop too)
DENY rm -rf build
DENY eval "rm -rf build"
#
# 1. $IFS read as a blank
DENY rm${IFS}-rf build
DENY rm$IFS-rf build
DENY rm${IFS}-rf${IFS}build
DENY rm$IFS$9-rf build
DENY rm${IFS:0:1}-rf build
DENY git${IFS}push${IFS}--force origin main
DENY bash -c 'rm${IFS}-rf build'
DENY ls; rm${IFS}-rf build
ALLOW while IFS= read -r line; do echo "$line"; done < f.txt
ALLOW OLDIFS=$IFS; IFS=,; echo a,b; IFS=$OLDIFS
ALLOW printf '%s' "$IFS" | od -c
ALLOW echo "rm${IFS}-rf build"
ALLOW grep 'rm${IFS}-rf' notes.txt
ALLOW rm${IFS}-v build
ALLOW IFS=: read -r a b <<< "x:y"; echo "$a"
# the view's text has an unbalanced quote: a structural deny inside a view only ends the view
ALLOW sh -c 'printf %s ${IFS:-"}" rm'
#
# 2. a verb held in a variable assigned earlier on the line. var-verb ships shadow: in a 55,000-command
# replay it denied one real command (`git -C $R checkout -- <file>`, a true discard develop missed).
# GH #375 now denies that command through the git rules, before any view runs.
WOULD_DENY var-verb X=rm; $X -rf build
WOULD_DENY var-verb X=rm && $X -rf build
# control: develop already denies a ${..} argv0 (it reads as every dangerous verb)
DENY X=rm; ${X} -rf build
WOULD_DENY var-verb X="rm -rf"; $X build
WOULD_DENY var-verb export X=rm; $X -rf build
WOULD_DENY var-verb X=git; $X push --force origin main
WOULD_DENY var-verb F=--force; git push $F origin main
WOULD_DENY var-verb A=-rf; rm $A build
WOULD_DENY var-verb X=mkfs.ext4; $X /dev/sda1
WOULD_DENY var-verb X=rm; bash -c "$X -rf build"
WOULD_DENY var-verb X=rm; echo hi; $X -rf build
WOULD_DENY var-verb X=rm; $X -rf build; X=ls
DENY R=/tmp/repo; git -C $R checkout -- notes.txt
ALLOW X=1; echo $X
ALLOW R=/tmp/x; RW=$R/w; git -C $RW remote add origin git@example.com:o/r.git
ALLOW X=rm; echo $X -rf build
ALLOW X=rm; "$X" -v build
ALLOW CMD=ls; $CMD -la
ALLOW D=build; ls $D && echo $D
ALLOW X="rm -rf"; echo "$X"
ALLOW N=3; for i in $(seq $N); do echo $i; done
ALLOW PY=python3; $PY -m pytest -q
#
# 3. a verb known only at run time: ask floor (shadow)
WOULD_ASK opaque-var-verb $X -rf build
WOULD_ASK opaque-var-verb "$RUNNER" build
# GH #593: a braced name is the same unknown program; a prefix assignment (`CMD=ls true`) sets the
# name only in that command's environment, so the shell still has none when `$CMD` runs
WOULD_ASK opaque-var-verb ${CMD} build
WOULD_ASK opaque-var-verb "${CMD}" build
WOULD_ASK opaque-var-verb ${CMD:-ls} build
WOULD_ASK opaque-var-verb ls && ${CMD} build
WOULD_ASK opaque-var-verb CMD=ls true; $CMD build
WOULD_ASK opaque-var-verb CMD=ls true && ${CMD} build
ALLOW CMD=ls; ${CMD} build
ALLOW CMD=ls CMD2=x; $CMD build
ALLOW export CMD=ls; $CMD build
ALLOW CMD="ls -la"; $CMD build
ALLOW CMD=$(command -v ls); $CMD build
ALLOW echo ${CMD} build
ALLOW ${HOME}/bin/tool build
WOULD_ASK source-file source ./setup.sh
WOULD_ASK source-file . ./setup.sh && git status
# GH #592: `.` with no candidate word on the line reaches python too (irrecoverable.sh fast path)
WOULD_ASK source-file . x.sh
WOULD_ASK source-file . ./x.sh
WOULD_ASK source-file command . x.sh
WOULD_ASK source-file ls && . x.sh
WOULD_ASK source-file ls; . x.sh
WOULD_ASK source-file (. x.sh)
WOULD_ASK source-file "." x.sh
WOULD_ASK source-file if true; then . x.sh; fi
WOULD_ASK source-file .<in x.sh
WOULD_ASK source-file .>/dev/null x.sh
WOULD_ASK source-file .&>/dev/null x.sh
WOULD_ASK source-file case a in a). x.sh;; esac
ALLOW cp -r . /tmp/x
ALLOW ls -la .
ALLOW echo done. next
ALLOW ./x.sh
ALLOW cd ..
#
# 4. mkfs, chmod 777 / a+rwx, rm --no-preserve-root
DENY mkfs.ext4 /dev/sda1
DENY mkfs -t ext4 /dev/sdb
DENY mkfs.vfat /dev/disk4s1
DENY sudo mkfs.ext4 /dev/sda1
DENY mke2fs /dev/sda1
DENY chmod -R 777 /
DENY chmod 777 /
DENY chmod 777 ~
DENY chmod -R 777 build
DENY chmod a+rwx /
DENY chmod -R a+rwx .
DENY chmod --recursive 0777 build
DENY chmod -vR 777 build
DENY sudo chmod -R 777 /usr/local
DENY rm -r --no-preserve-root /
DENY rm --no-preserve-root -r /tmp/x
DENY rm --no-preserve-r -r /
# the 777 permission set in any letter order, with X, or granted to u, g and o clause by clause
DENY chmod -R a+rwX ./x
DENY chmod -R a+wrx ./x
DENY chmod -R ugo=xwr ./x
DENY chmod a+Xrw /
DENY chmod -R u+rwx,g+rwx,o+rwx ./x
DENY chmod -R u=rwX,go+rwX ./x
ALLOW chmod -R u+rwX ./x
ALLOW chmod a+r ./x
ALLOW chmod -R a+rX ./x
ALLOW chmod -R go+rX ./x
ALLOW chmod -R u+rwx,g+rx,o+rx ./x
ALLOW chmod 644 a+wrx
# a removal never lets a mode through: the gate ignores - clauses on purpose (see the comment at _CHMOD_CLAUSE)
DENY chmod -R a+rwx,a-w,a+w ./x
DENY chmod -R a-w,a+rwx ./x
DENY chmod -R a=r,a+rwx ./x
DENY chmod -R u+rwx,g+rwx,o+rwx,o-x,o+x ./x
DENY chmod -R a+rwx,o-x,+x ./x
DENY chmod -R a+rwx,o-w,o=u ./x
DENY chmod -R a+rwx,o-w$(echo ,o+w) ./x
DENY chmod -R a+rwx,o-w ./x
DENY V=a+rwx; chmod -R $V a+rwx,o-w ./x
# GNU rm takes an unambiguous long-option prefix: --r only names --recursive, --f only --force
DENY rm --rec --force ./build
DENY rm -r --forc ./build
DENY rm --r --f ./build
DENY rm --recursive --fo ./build
ALLOW rm --dir --force ./build
# rm -r alone denies too (rule rm-recursive, 2026-10-25), so these prefix rows deny whatever the long option is
DENY rm -r --forcex ./build
DENY rm --rec --verbose ./build
ALLOW rm --dir --forcex ./build
DENY rm --verbose --re ./build
ALLOW rm --verbose --dir ./build
DENY echo `date` $(rm --rec --forc ./build)
ALLOW echo `date` $(rm --verbose ./build)
ALLOW mkfs.ext4 -V
ALLOW echo mkfs.ext4 /dev/sda1
ALLOW chmod +x run.sh
ALLOW chmod 755 script.sh
ALLOW chmod -R 755 build
ALLOW chmod 777 /tmp/scratch.sock
ALLOW chmod -R u+w build
ALLOW chmod 644 ~/.ssh/config
ALLOW chmod 600 ~/.netrc
ALLOW git diff -- chmod.txt
ALLOW rm --preserve-root -i x
