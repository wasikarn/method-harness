# GH #375 cases for hooks/gates/irrecoverable.sh, replayed by tests/hooks/test-gates-other.sh.
# DENY <cmd>: exit 2. ALLOW <cmd>: exit 0, empty stdout, no journal row.
# shlex splits one shell word at a character outside its word characters ($, :, @, a Thai letter),
# so a git global's value (-C $R) took the variable's name for the subcommand. Each DENY shape in 1-5 was
# run in /bin/bash 3.2, bash 5, /bin/dash and /bin/zsh with the variables set, against a logging git stub,
# and ran a destructive git argv in each.
#
# 1. an unquoted variable in a -C value, spaced or glued
DENY git -C $PWD reset --hard
DENY git -C $HOME/x push --force origin main
DENY git -C $R clean -fd
DENY git -C $R checkout -- f.txt
DENY git -C$R reset --hard
DENY git -C $A$B reset --hard
DENY git -C ~/$R reset --hard
DENY git -C $R/sub restore f.txt
DENY git -C "$R"$S reset --hard
DENY git -C $(pwd)/$R reset --hard
DENY git -C ${R}x$S reset --hard
DENY git -C $1 reset --hard
DENY git --no-pager -C $R push -f origin main
DENY R=/tmp/r; git -C $R checkout -- f.txt
# 2. every destructive git rule after the variable
DENY git -C $R branch -D main
DENY git -C $R stash drop
DENY git -C $R stash clear
DENY git -C $R switch -f main
DENY git -C $R commit --amend -m x
DENY git -C $R commit -n -m x
DENY git -C $R add -A
DENY git -C $R push origin +main
# 3. the other value-taking globals
DENY git --git-dir=$G reset --hard
DENY git --git-dir $G/.git --work-tree $W reset --hard
DENY git --work-tree=$W checkout .
DENY git --namespace=$N push --force origin main
DENY git -c user.name=$N reset --hard
DENY git -c core.hooksPath=$H commit -m x
DENY git -ccore.hooksPath=$H commit -m x
DENY git --config-env=core.hooksPath=$H commit -m x
# an empty value unwires every hook (git 2.43, live-checked), so it denies like any other; was an ALLOW control
DENY git -c core.hooksPath= status
# 4. other characters shlex splits a word at
DENY git -c user.email=a@b.c commit --amend -m x
DENY git -C a:b reset --hard
DENY git -C ~/โปรเจกต์ reset --hard
DENY git -C a$ reset --hard
DENY git -C ~/my\ $R reset --hard
# 5. the same through a wrapper
DENY sudo git -C $R reset --hard
DENY env A=1 git -C $R reset --hard
DENY echo f | xargs git -C $R reset --hard
DENY find . -maxdepth 0 -exec git -C $R reset --hard \;
# 6. an unquoted -C value now reads like the quoted one ("$R"). develop denies `git checkout $BR`:
# bash and dash split BR="HEAD f.txt" into a tree-ish and a path.
DENY git -C $R checkout $BR
# 7. the token reading stays beside the word reading, so no develop deny is lost (not destructive:
# with reset unset, git gets `-C --hard`)
DENY git -C $reset --hard
#
# controls
ALLOW git -C $REPO status
ALLOW git -C "$d" log
ALLOW git -C $x diff
ALLOW git -C $R log --oneline -5
ALLOW git -C $HOME/x fetch origin
ALLOW git -C $R push origin main
ALLOW git -C $R checkout -b feat
ALLOW git -C $R reset HEAD~1
ALLOW git -C $R stash list
ALLOW git -C $R commit -m "reset --hard"
ALLOW git --git-dir=$G rev-parse HEAD
ALLOW git -c user.email=a@b.c commit -m x
ALLOW git -C ~/โปรเจกต์ status
ALLOW git -C a:b diff
ALLOW git -C $ status
ALLOW for d in */; do git -C $d status --short; done
